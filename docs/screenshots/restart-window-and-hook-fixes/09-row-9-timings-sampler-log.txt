Row 9 timings: the in-page sampler's log, 2026-11-07, tab-7 (chat with hooks, Claude Code 2.1.319 on Haiku)

How it was taken: the sampler below was pasted into the page through `drive eval` just before each prompt was
sent, then the restart window was opened. Every 100 ms it read tab-7's record status and notification
(window.__afterterm.tab), whether the last 25 lines of tab-7's buffer held dialog text (DIALOG), how many restart
window rows read "Stops mid-turn" (every other thread was idle, so any such row was tab-7's), and the footer's
data-status ("window-closed" before the window opened). It logged a line only when one of those changed. Times are
seconds from the moment the sampler was pasted in.

The sampler:

(() => {
  if (window.__r9) clearInterval(window.__r9.timer);
  const r = { t0: Date.now(), log: [], last: '', snaps: [] };
  r.timer = setInterval(() => {
    const t = window.__afterterm.tab('tab-7');
    let tail = window.__afterterm.tail('tab-7', 25) || [];
    if (Array.isArray(tail)) tail = tail.join('\n');
    const dlg = /Do you want|Enter to select|Esc to cancel|Enter to confirm/.test(tail) ? 'DIALOG' : 'no-dialog';
    const rows = [...document.querySelectorAll('[data-restart-rows] .tr')];
    const mid = rows.filter(x => x.dataset.outcome === 'stops-mid-turn').length;
    const st = document.querySelector('[data-status]')?.dataset.status ?? 'window-closed';
    const key = ['record=' + (t?.recordStatus ?? '-'), 'notif=' + (t?.notification ?? '-'), dlg, 'midTurnRows=' + mid, 'footer=' + st].join(' | ');
    if (key !== r.last) { r.last = key; r.log.push(((Date.now() - r.t0) / 1000).toFixed(1) + 's ' + key); if (dlg === 'DIALOG' && r.snaps.length < 3) r.snaps.push(tail.split('\n').slice(-14).join('\n')); }
  }, 100);
  window.__r9 = r; return 'r9 on';
})()
